import numpy as np
from scipy.stats import norm

def calculate_log_likelihoods(data):
    # Extract key numerical values
    temp_increase_since_1880 = data["numerical_values"][0]["value"]
    temp_increase_uncertainty = data["numerical_values"][0].get("uncertainty", None)
    
    temp_rate_since_1975 = data["numerical_values"][1]["value"]
    temp_rate_range = data["numerical_values"][1].get("range", None)

    base_period_temp = data["numerical_values"][3]["value"]
    base_temp_uncertainty = data["numerical_values"][3].get("uncertainty", None)

    confidence_score = data["confidence_assessment"]["confidence_score"]

    # Handling uncertainties and establishing reasonable priors
    temp_increase_prior_sigma = 0.2 if temp_increase_uncertainty == "at least" else 0.1
    temp_rate_sigma = (temp_rate_range[1] - temp_rate_range[0]) / 4 if temp_rate_range else 0.05
    base_temp_sigma = 0.3 if base_temp_uncertainty else 0.1

    # Assume hypotheses:
    # H+: Human-generated GHG emissions significantly increase global temperatures.
    # H-: Natural factors alone drive temperature changes.

    # Define synthetic likelihoods based on expected distributions
    # Assume under H+: Observed warming should be high
    mu_plus_temp_increase = 1.0
    mu_plus_temp_rate = 0.18  

    # Assume under H-: Less or no significant warming
    mu_minus_temp_increase = 0.2
    mu_minus_temp_rate = 0.02  

    # Compute log-likelihoods using normal distributions
    l_plus = (
        np.log(norm.pdf(temp_increase_since_1880, mu_plus_temp_increase, temp_increase_prior_sigma)) +
        np.log(norm.pdf(temp_rate_since_1975, mu_plus_temp_rate, temp_rate_sigma))
    )
    
    l_minus = (
        np.log(norm.pdf(temp_increase_since_1880, mu_minus_temp_increase, temp_increase_prior_sigma)) +
        np.log(norm.pdf(temp_rate_since_1975, mu_minus_temp_rate, temp_rate_sigma))
    )

    # Apply confidence weighting
    confidence_weight = np.clip(confidence_score, 0.5, 0.95)
    l_plus *= confidence_weight
    l_minus *= confidence_weight

    # Cap log-likelihood differences to avoid overconfidence
    max_diff = 5.0
    log_likelihood_diff = np.clip(l_plus - l_minus, -max_diff, max_diff)
    
    # Adjust likelihoods accordingly
    l_plus = l_minus + log_likelihood_diff

    # Print intermediate values
    print(f"Temperature increase since 1880: {temp_increase_since_1880} ± {temp_increase_prior_sigma}")
    print(f"Temperature rate since 1975: {temp_rate_since_1975} ± {temp_rate_sigma}")
    print(f"log P(data | H+): {l_plus:.4f}")
    print(f"log P(data | H-): {l_minus:.4f}")

    # Ensure numerical stability (avoid NaN or inf)
    if np.isnan(l_plus) or np.isinf(l_plus):
        l_plus = -np.inf
    if np.isnan(l_minus) or np.isinf(l_minus):
        l_minus = -np.inf

    print(f"Final log-likelihoods: (l_plus={l_plus}, l_minus={l_minus})")

    return l_plus, l_minus

import numpy as np
from scipy.stats import norm

def calculate_log_likelihoods(data_dict):
    # Extract relevant data
    confidence_score = data_dict["confidence_assessment"]["confidence_score"]
    key_limitations = data_dict["confidence_assessment"]["key_limitations"]

    # Assign base likelihoods
    base_likelihood_hypothesis = confidence_score  # As given, 0.9
    base_likelihood_not_hypothesis = 1 - confidence_score  # 1 - 0.9 = 0.1

    # Handle uncertainties and potential biases
    uncertainty_penalty = 0.1 * len(key_limitations)  # Penalize if there are more uncertainties
    adjusted_likelihood_hypothesis = max(0.05, base_likelihood_hypothesis - uncertainty_penalty)
    adjusted_likelihood_not_hypothesis = max(0.05, base_likelihood_not_hypothesis + uncertainty_penalty)

    # Normalize to avoid values summing above 1
    total = adjusted_likelihood_hypothesis + adjusted_likelihood_not_hypothesis
    adjusted_likelihood_hypothesis /= total
    adjusted_likelihood_not_hypothesis /= total

    # Calculate log likelihoods
    l_plus = np.log(adjusted_likelihood_hypothesis)
    l_minus = np.log(adjusted_likelihood_not_hypothesis)

    # Cap absolute difference
    max_diff = 5.0
    actual_diff = l_plus - l_minus
    if abs(actual_diff) > max_diff:
        adjustment = (abs(actual_diff) - max_diff) / 2
        l_plus -= adjustment * np.sign(actual_diff)
        l_minus += adjustment * np.sign(actual_diff)

    # Ensure numerical stability
    l_plus = np.clip(l_plus, -20, 20)
    l_minus = np.clip(l_minus, -20, 20)

    # Print intermediate results
    print(f"Base likelihood for hypothesis: {base_likelihood_hypothesis}")
    print(f"Base likelihood for not hypothesis: {base_likelihood_not_hypothesis}")
    print(f"Adjusted likelihood for hypothesis: {adjusted_likelihood_hypothesis}")
    print(f"Adjusted likelihood for not hypothesis: {adjusted_likelihood_not_hypothesis}")
    print(f"Log likelihood for hypothesis (l_plus): {l_plus}")
    print(f"Log likelihood for not hypothesis (l_minus): {l_minus}")

    return l_plus, l_minus
